```html
<!doctype html>
<html>
  <head>
    <title>Lumememm</title>
    <script>
      function joonista(){

        // 1. Alumine pall (suur)
        g.beginPath();
        g.arc(200, 220, 50, 0, 2 * Math.PI);
        g.stroke();

        // 2. Keskmine pall
        g.beginPath();
        g.arc(200, 140, 35, 0, 2 * Math.PI);
        g.stroke();

        // 3. Ülemine pall (pea)
        g.beginPath();
        g.arc(200, 80, 25, 0, 2 * Math.PI);
        g.stroke();

        // Parem silm

        g.fillStyle = "orange";
        g.beginPath();
        g.moveTo(200, 80);
        g.lineTo(218, 83);
        g.lineTo(200, 86);
        g.fill();
      }
    </script>
  </head>
  <body onload="joonista()">
```

### Kuidas lahendus töötab:

1. **Funktsioon `joonistaLumememm(x, y, d)`** (või lühemalt `lumememm(x, y, d)`):
   - **Parameetrid**:
     - `x`: alumise palli keskpunkti x-koordinaat
     - `y`: alumise palli keskpunkti y-koordinaat
     - `d`: alumise ringi läbimõõt
   - **Dünaamiline skaleerimine**:
     - Alumise palli raadius: $r_1 = d / 2$
     - Keskmise palli raadius: $r_2 = r_1 \times 0.7$
     - Pea raadius: $r_3 = r_1 \times 0.5$
     - Silmad ja nina skaleeruvad automaatselt proportsioonis pea suurusega.
2. **Kahe lumememme demo (`joonista()`)**:
   - Esimene (suurem) lumememm: `joonistaLumememm(130, 220, 90)`
   - Teine (väiksem) lumememm: `joonistaLumememm(270, 235, 60)`

### Faili `joonis4.html` sisu:

```html
<!doctype html>
<html>
  <head>
    <title>Lumememmede joonistamine</title>
    <script>
      /**
       * Funktsioon lumememme joonistamiseks lõuendile.
       * @param {number} x - Alumise palli keskpunkti x-koordinaat
       * @param {number} y - Alumise palli keskpunkti y-koordinaat
       * @param {number} d - Alumise ringi läbimõõt
       * @param {CanvasRenderingContext2D} [g] - 2D graafikakontekst (valikuline)
       */
      function joonistaLumememm(x, y, d, g) {
        if (!g) {
          g = document.getElementById("tahvel").getContext("2d");
        }

        // Raadiused proportsioonis alumise ringi läbimõõduga d
        let r1 = d / 2;       // Alumise palli raadius
        let r2 = r1 * 0.7;    // Keskmise palli raadius
        let r3 = r1 * 0.5;    // Pea raadius

        // Vertikaalsed keskpunktid (väikese katvusega, et pallid toetuksid loomulikult)
        let y1 = y;
        let y2 = y1 - r1 - r2 * 0.85;
        let y3 = y2 - r2 - r3 * 0.85;

        // 1. Alumine pall (suur)
        g.beginPath();
        g.arc(x, y1, r1, 0, 2 * Math.PI);
        g.stroke();

        // 2. Keskmine pall
        g.beginPath();
        g.arc(x, y2, r2, 0, 2 * Math.PI);
        g.stroke();

        // 3. Ülemine pall (pea)
        g.beginPath();
        g.arc(x, y3, r3, 0, 2 * Math.PI);
        g.stroke();

        // 4. Silmad
        let eyeRadius = Math.max(1.5, r3 * 0.12);
        let eyeOffsetX = r3 * 0.35;
        let eyeOffsetY = r3 * 0.2;

        g.fillStyle = "black";

        // Vasak silm
        g.beginPath();
        g.arc(x - eyeOffsetX, y3 - eyeOffsetY, eyeRadius, 0, 2 * Math.PI);
        g.fill();

        // Parem silm
        g.beginPath();
        g.arc(x + eyeOffsetX, y3 - eyeOffsetY, eyeRadius, 0, 2 * Math.PI);
        g.fill();

        // 5. Nina (porgand)
        g.fillStyle = "orange";
        g.beginPath();
        g.moveTo(x, y3);
        g.lineTo(x + r3 * 0.7, y3 + r3 * 0.12);
        g.lineTo(x, y3 + r3 * 0.24);
        g.closePath();
        g.fill();
      }

      // Sünonüüm lühema nime jaoks
      const lumememm = joonistaLumememm;

      // Demo kahe lumememme joonistamiseks
      function joonista() {
        let g = document.getElementById("tahvel").getContext("2d");
        g.clearRect(0, 0, 400, 300);

        // Esimene lumememm (suurem): x=130, y=220, läbimõõt d=90
        joonistaLumememm(130, 220, 90, g);

        // Teine lumememm (väiksem): x=270, y=235, läbimõõt d=60
        joonistaLumememm(270, 235, 60, g);
      }
    </script>
  </head>
  <body onload="joonista()">
    <h1>Kahe lumememme demo</h1>
    <p>Lumememmed on joonistatud funktsiooni <code>joonistaLumememm(x, y, d)</code> abil.</p>
    <canvas id="tahvel" width="400" height="300" style="border: 1px solid #ccc;">
    </canvas>
  </body>
</html>
```

   ```javascript
        { kuupaev: "13.11.2026", temp: 4, tuul: 4, puhang: 12 },
   ```

```html
       * Funktsioon ilmalumememme joonistamiseks.
       * @param {number} x - Alumise palli keskpunkti x-koordinaat
       * @param {number} y - Alumise palli keskpunkti y-koordinaat
       * @param {number} temperatuur - Õhutemperatuur (°C, määrab alumise palli suuruse)
       * @param {number} tuul - Tuule kiirus (m/s, määrab keskmise palli suuruse)
       * @param {number} puhang - Puhangukiirus (m/s, määrab ülemise palli suuruse)
       * @param {string} kuupaev - Kuupäev, mis joonistatakse alumise palli keskele
       * @param {CanvasRenderingContext2D} [g] - 2D graafikakontekst (valikuline)
       */
      function joonistaIlmaLumememm(x, y, temperatuur, tuul, puhang, kuupaev, g) {
        if (!g) {
          g = document.getElementById("tahvel").getContext("2d");
        }

        // 1. Pallide raadiused vastavalt ilmanäitajatele
        let r1 = 15 + temperatuur * 5;  // Alumine pall (temperatuur)
        let r2 = 10 + tuul * 3;          // Keskmine pall (tuule kiirus)
        let r3 = 6 + puhang * 1.1;       // Ülemine pall (puhangukiirus)

        // Vertikaalsed keskpunktid
        let y1 = y;
        let y2 = y1 - r1 - r2 * 0.85;
        let y3 = y2 - r2 - r3 * 0.85;

        // 1. Alumine pall (temperatuur)
        g.strokeStyle = "black";
        g.lineWidth = 1.5;
        g.beginPath();
        g.arc(x, y1, r1, 0, 2 * Math.PI);
        g.stroke();

        // Kuupäev alumise palli keskele

        g.font = "bold 12px Arial";
        g.textAlign = "center";
        g.textBaseline = "middle";
        g.fillText(kuupaev, x, y1);

        // 2. Keskmine pall (tuule kiirus)
        g.beginPath();
        g.arc(x, y2, r2, 0, 2 * Math.PI);
        g.stroke();

        // 3. Ülemine pall / pea (puhangukiirus)
        g.beginPath();
        g.arc(x, y3, r3, 0, 2 * Math.PI);
        g.stroke();

        // 4. Silmad ülemisel pallil

        g.fill();

        g.fill();
```
